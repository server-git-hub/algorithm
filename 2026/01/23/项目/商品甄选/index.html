package com.spzx.product.service.impl;

import com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper;

/**

  • 商品规格 服务实现类

  • @author atguigu

  • @since 2026-02-01
    */
    @Service
    public class ProductSpecServiceImpl extends ServiceImpl<ProductSpecMapper, ProductSpec> implements IProductSpecService {

    @Autowired
    private ProductSpecMapper productSpecMapper;

    @Override
    public List selectProductSpecList() {
    return productSpecMapper.selectAll();
    }

    @Override
    public void saveProductSpec(ProductSpec productSpec){
    String specValue = productSpec.getSpecValue();
    validateSpecJson(specValue);
    productSpec.setCreateBy(SecurityUtils.getUsername());
    productSpec.setUpdateBy(SecurityUtils.getUsername());
    this.save(productSpec);
    }

    @Override
    public List queryProductSpecByCategoryId(Long id) {
    List productSpecs = productSpecMapper.selectList(
    new LambdaQueryWrapper()
    .eq(ProductSpec::getCategoryId, id));
    return productSpecs;
    }

    private static final ObjectMapper MAPPER = new ObjectMapper();

    public static void validateSpecJson(String specValueJson) {
    try {
    JsonNode root = MAPPER.readTree(specValueJson);

    if (root == null || !root.isArray()) {
    throw new ServiceException(“规格格式错误：应为 JSON 数组”);
    }

    // 校验 key 全局唯一
    Set keySet = new HashSet<>();

    for (JsonNode item : root) {
    if (item == null || !item.isObject()) {
    throw new ServiceException(“规格格式错误：数组元素必须是对象”);
    }

    String key = item.path(“key”).asText(“”).trim();
    if (key.isEmpty()) {
    throw new ServiceException(“规格名称为空”);
    }
    if (!keySet.add(key)) {
    throw new ServiceException(“规格名称重复：” + key);
    }

    JsonNode valueListNode = item.path(“valueList”);
    if (!valueListNode.isArray() || valueListNode.size() == 0) {
    throw new ServiceException(“规格值为空：” + key);
    }

    // 校验当前 key 的 valueList 唯一
    Set valueSet = new HashSet<>();
    for (JsonNode v : valueListNode) {
    String val = v.asText(“”).trim();
    if (val.isEmpty()) {
    throw new ServiceException(“规格值为空：” + key);
    }
    if (!valueSet.add(val)) {
    throw new ServiceException(“规格值重复：” + key + “ = “ + val);
    }
    }
    }
    } catch (ServiceException e) {
    throw e;
    } catch (Exception e) {
    // JSON 解析失败等
    throw new ServiceException(“规格 JSON 解析失败：” + e.getMessage());
    }
    }

}

    @Service

    @Override

    }

    @Override
    @Transactional(rollbackFor = Exception.class)
    public void saveProduct(Product product) {
    List productSkus = product.getProductSkuList();
    List detailsImageUrlList = product.getDetailsImageUrlList();
    //存储product 商品表
    if(!ObjectUtils.isEmpty(product)){
    product.setStatus(0); //设置默认装填
    product.setCreateBy(SecurityUtils.getUsername()); //设置创建者
    productMapper.insert(product);
    }
    //存储productsku 商品库存单位表
    if(!ObjectUtils.isEmpty(productSkus)){
    int num=1;
    for (ProductSku skus : productSkus) {
    skus.setStatus(0); //设置初始状态
    skus.setCreateBy(SecurityUtils.getUsername()); //设置创建者
    skus.setProductId(product.getId()); //设置所属商品id
    skus.setSkuCode(product.getId()+”_”+(num++));
    skus.setSkuName(product.getName()+””+skus.getSkuSpec());
    productSkuMapper.insert(skus);
    SkuStock skuStock = new SkuStock();
    skuStock.setSkuId(skus.getId());
    skuStock.setTotalNum(skus.getStockNum());
    skuStock.setLockNum(0);
    skuStock.setAvailableNum(skus.getStockNum());
    skuStock.setSaleNum(0);
    skuStock.setStatus(0);
    skuStock.setCreateBy(SecurityUtils.getUsername());
    skuStockMapper.insert(skuStock);
    }
    }
    //存储productdetails 商品细节表
    if(!ObjectUtils.isEmpty(detailsImageUrlList)){
    ProductDetails productDetails = new ProductDetails();
    productDetails.setProductId(product.getId());
    productDetails.setCreateBy(SecurityUtils.getUsername());
    String join = String.join(“,”, detailsImageUrlList);
    productDetails.setImageUrls(join);
    productDetailsMapper.insert(productDetails);
    }
    }

    @Override
    @Transactional(rollbackFor = Exception.class)
    public void updateStatus(Long spuId, Integer status) {
    //变更商品状态
    LambdaUpdateWrapper productLambdaUpdateWrapper = new LambdaUpdateWrapper<>();
    productLambdaUpdateWrapper.eq(Product::getId, spuId);
    productLambdaUpdateWrapper.set(Product::getStatus, status);
    productLambdaUpdateWrapper.set(Product::getUpdateBy, SecurityUtils.getUsername());
    productMapper.update(null, productLambdaUpdateWrapper);
    //变更sku状态
    LambdaUpdateWrapper productSkuLambdaUpdateWrapper = new LambdaUpdateWrapper<>();
    productSkuLambdaUpdateWrapper.eq(ProductSku::getProductId,spuId);
    productSkuLambdaUpdateWrapper.set(ProductSku::getStatus, status);
    productSkuLambdaUpdateWrapper.set(ProductSku::getUpdateBy, SecurityUtils.getUsername());
    productSkuMapper.update(null, productSkuLambdaUpdateWrapper);
    //变更库存状态
    LambdaQueryWrapper productSkuLambdaQueryWrapper = new LambdaQueryWrapper<>();
    productSkuLambdaQueryWrapper.eq(ProductSku::getProductId,spuId);
    productSkuLambdaQueryWrapper.select(ProductSku::getId);
    List skuIds = productSkuMapper.selectList(productSkuLambdaQueryWrapper)
    .stream()
    .map(ProductSku::getId)
    .collect(Collectors.toList());
    if(!ObjectUtils.isEmpty(skuIds)){
    LambdaUpdateWrapper skuStockLambdaUpdateWrapper = new LambdaUpdateWrapper<>();
    skuStockLambdaUpdateWrapper.in(SkuStock::getSkuId, skuIds);
    skuStockLambdaUpdateWrapper.set(SkuStock::getStatus, status);
    skuStockLambdaUpdateWrapper.set(SkuStock::getUpdateBy, SecurityUtils.getUsername());
    skuStockMapper.update(null, skuStockLambdaUpdateWrapper);
    }
    }

    @Override
    @Transactional(rollbackFor = Exception.class)
    public void deleteStatus(List ids) {
    //1删除Product
    productMapper.deleteBatchIds(ids);
    //2查询SkuId
    List productSkus = productSkuMapper.selectList(
    new LambdaQueryWrapper()
    .in(ProductSku::getProductId, ids)
    .select(ProductSku::getId));
    if(!ObjectUtils.isEmpty(productSkus)){
    List skuIds = productSkus
    .stream()
    .map(ProductSku::getId)
    .collect(Collectors.toList());
    //3删除ProductSku
    productSkuMapper.deleteBatchIds(skuIds);
    //4删除StuStock
    skuStockMapper.delete(new LambdaQueryWrapper().in(SkuStock::getSkuId,skuIds));
    }
    //5删除ProductDetails
    LambdaQueryWrapper productDetailsLambdaQueryWrapper = new LambdaQueryWrapper<>();
    productDetailsLambdaQueryWrapper.in(ProductDetails::getProductId,ids);
    productDetailsMapper.delete(productDetailsLambdaQueryWrapper);
    }
